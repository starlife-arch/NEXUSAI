async function signInWithGoogle() {
  try {
    const provider = new firebase.auth.GoogleAuthProvider();
    await window.firebaseAuth.signInWithPopup(provider);
    closeModal();
    state.currentView = 'user-chat';
    render();
    showToast('Signed in successfully with Google!', 'success', 'Welcome back');
  } catch (error) {
    console.error('Google sign-in failed:', error);
    showToast(error.message || 'Google sign-in failed. Please try again.', 'error', 'Sign-in failed');
  }
}

async function signInWithEmail() {
  const nameInput = document.getElementById('auth-name');
  const emailInput = document.getElementById('auth-email');
  const passwordInput = document.getElementById('auth-password');
  const email = emailInput ? emailInput.value.trim() : '';
  const name = nameInput ? nameInput.value.trim() : '';
  const password = passwordInput ? passwordInput.value : '';

  if (!email || !password) {
    showToast('Please enter your email and password.', 'error', 'Missing information');
    return;
  }
  if (password.length < 6) {
    showToast('Password must be at least 6 characters.', 'error', 'Invalid password');
    return;
  }

  try {
    if (state.authMode === 'signup') {
      if (!name) {
        showToast('Please enter your name.', 'error', 'Missing name');
        return;
      }
      const credential = await window.firebaseAuth.createUserWithEmailAndPassword(email, password);
      await credential.user.updateProfile({ displayName: name });
    } else {
      await window.firebaseAuth.signInWithEmailAndPassword(email, password);
    }
    closeModal();
    state.currentView = 'user-chat';
    render();
    showToast(state.authMode === 'signup' ? 'Account created successfully.' : 'Welcome back!', 'success');
  } catch (error) {
    console.error('Email authentication failed:', error);
    const messages = {
      'auth/email-already-in-use': 'An account already exists for this email.',
      'auth/invalid-email': 'Please check the email address.',
      'auth/invalid-credential': 'Email or password is incorrect.',
      'auth/weak-password': 'Choose a stronger password (at least 6 characters).',
      'auth/popup-closed-by-user': 'The Google sign-in window was closed.'
    };
    showToast(messages[error.code] || error.message || 'Authentication failed.', 'error', 'Sign-in failed');
  }
}

async function sendMessage() {
  const input = document.getElementById('user-message-input');
  if (!input || !input.value.trim()) return;

  const user = window.firebaseAuth.currentUser;
  if (!user) {
    showToast('Please sign in before chatting with AI.', 'warning');
    openAuthModal('signin');
    return;
  }

  const content = input.value.trim();
  mockMessages.push({
    id: mockMessages.length + 1,
    role: 'user',
    content,
    timestamp: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
  });
  state.newMessage = '';
  state.isTyping = true;
  state.messagesToday++;
  render();

  try {
    const token = await user.getIdToken();
    const response = await fetch('/.netlify/functions/chat', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer ' + token
      },
      body: JSON.stringify({
        messages: mockMessages.slice(-20).map((message) => ({
          role: message.role === 'ai' ? 'assistant' : message.role,
          content: message.content
        }))
      })
    });
    const data = await response.json().catch(() => ({}));
    if (!response.ok) throw new Error(data.error || 'The AI request failed.');

    mockMessages.push({
      id: mockMessages.length + 1,
      role: 'ai',
      content: data.reply,
      timestamp: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
    });
    state.apiCalls++;
  } catch (error) {
    mockMessages.push({
      id: mockMessages.length + 1,
      role: 'ai',
      content: 'Sorry, I could not get an AI response. ' + (error.message || 'Please try again.'),
      timestamp: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
    });
  } finally {
    state.isTyping = false;
    render();
    const el = document.getElementById('chat-messages');
    if (el) el.scrollTop = el.scrollHeight;
  }
}

window.firebaseAuth.onAuthStateChanged((user) => {
  if (user) {
    state.userName = user.displayName || user.email?.split('@')[0] || 'User';
    state.userEmail = user.email || '';
    state.userAvatar = state.userName.charAt(0).toUpperCase();
    state.userProvider = user.providerData?.[0]?.providerId === 'google.com' ? 'Google' : 'Email';
    state.memberSince = state.memberSince || new Date().toLocaleDateString('en-US', { month: 'short', year: 'numeric' });
    if (state.currentView === 'landing') state.currentView = 'user-chat';
  } else {
    state.userName = '';
    state.userEmail = '';
    state.userAvatar = '';
    state.userProvider = '';
    state.currentRole = 'user';
    if (state.currentView !== 'admin-login') state.currentView = 'landing';
  }
  render();
});

G. Lock the fake admin login
----------------------------
The current app's admin login is not secure: it contains hard-coded demo credentials and grants admin access without checking a password.
Do NOT leave it as-is for a public site.

At minimum, replace the current function:
  function loginAdmin() { state.currentRole = 'admin'; ... }

with:

async function loginAdmin() {
  const email = document.getElementById('admin-email')?.value.trim();
  const password = document.getElementById('admin-password')?.value;
  if (!email || !password) {
    showToast('Enter your admin email and password.', 'error');
    return;
  }
  try {
    const credential = await window.firebaseAuth.signInWithEmailAndPassword(email, password);
    const token = await credential.user.getIdTokenResult(true);
    if (token.claims.admin !== true) {
      state.currentRole = 'user';
      state.currentView = 'landing';
      render();
      showToast('This account does not have administrator access.', 'error', 'Access denied');
      return;
    }
    state.currentRole = 'admin';
    state.currentView = 'admin-dashboard';
    state.adminTab = 'dashboard';
    render();
    showToast('Admin access verified.', 'success');
  } catch (error) {
    showToast('Admin sign-in failed. Check your credentials.', 'error');
  }
}

Also, inside function renderAdminLogin(), remove the pre-filled value attributes:
  value="admin@laud.com"
  value="admin123"

and add id="admin-email" to the email input and id="admin-password" to the password input.
This will deny admin access unless a trusted administrator has set the Firebase custom claim `admin: true` using a secure server-side Firebase Admin SDK process. Do not set admin claims from browser code.

H. Configure Netlify secret
---------------------------
In Netlify dashboard for https://drovix.netlify.app:
Project configuration → Environment variables → Add a variable
Key: GROQ_API_KEY
Value: your Groq API key (paste it in Netlify only; never in GitHub or this chat)
Scope: Functions, if Netlify offers scope selection.
Optionally add GROQ_MODEL = openai/gpt-oss-20b.
Save, then trigger a fresh deploy. Netlify only applies environment-variable changes to a new deploy.

I. Enable Firebase providers
----------------------------
Firebase Console → project `drovix` → Authentication → Get started.
Sign-in method: enable Email/Password and Google.
Authentication → Settings → Authorized domains: add `drovix.netlify.app`.
Do not disable Firebase App Check/security protections if you already use them.

Notes:
- Firebase's web config is intended to be used by browser apps; it is not a service-account credential. Keep Google Cloud API restrictions appropriate for your site.
- The Groq secret is never sent to the browser; the Netlify function uses it on the server.
- The current app still keeps chat messages in browser memory. Firestore has been created but chat/profile persistence and Firestore security rules are a separate follow-up task.
- Admin login will be denied until you configure a custom admin claim through a trusted server-side process.
